# LivingroomOS – required Debian 13 packages ("Trixie"; amd64).
# Installation:  sudo apt install --no-install-recommends $(grep -v '^[[:space:]]*#' debian-packages.txt | awk '{print $1}')
# Package rationale: docs/SYSTEM_DEPENDENCIES.md. Optional packages: debian-packages-optional.txt.
# One package per line, followed by an optional comment.
python3                # Backend runtime (Debian 13: 3.13; version >= 3.11 required)
python3-fastapi        # Backend web framework (installs starlette and pydantic as dependencies)
python3-uvicorn        # ASGI-Server
labwc                  # Wayland-Compositor
greetd                 # Automatic login to the labwc session
systemd                # Services and user units (already included with Debian)
dbus-user-session      # starts systemd --user for each session
libpam-systemd         # creates the logind session (GPU and input-device access)
polkitd                # authorization for restricted restart/shutdown actions
pipewire               # Audio
pipewire-pulse         # audio compatibility for Google Chrome
wireplumber            # PipeWire session management
libgl1-mesa-dri        # Mesa-Grafiktreiber (Intel/AMD x86_64)
fonts-noto-core        # fonts for the interface and websites
fonts-noto-color-emoji # Emoji in Webseiten
